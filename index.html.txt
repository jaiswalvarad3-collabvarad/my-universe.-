<script>
const $ = id => document.getElementById(id);

let villains = [];
let extras = [];
let history = [];
let temporary = false;
let savedTemporaryState = null;

const themes = {
  fantasy: {
    sky:"🌙", objects:"🏰　💎　🌲",
    background:"linear-gradient(#514bb0,#b17bd1 58%,#ffcfb0)",
    land:"#303c62", ground:"#26324b"
  },
  space: {
    sky:"🪐", objects:"🚀　🌟　🛸",
    background:"linear-gradient(#03051b,#20266b 60%,#543a8b)",
    land:"#282747", ground:"#11142b"
  },
  forest: {
    sky:"☀️", objects:"🌲　🦋　🍄",
    background:"linear-gradient(#79c7d7,#c8ebad 60%,#e2d8a3)",
    land:"#376d42", ground:"#254d32"
  },
  ocean: {
    sky:"🐬", objects:"🐠　🪸　🐚",
    background:"linear-gradient(#1ac2e4,#0876b6 65%,#07456e)",
    land:"#126b81", ground:"#084358"
  },
  lava: {
    sky:"🌋", objects:"🔥　🪨　🐉",
    background:"linear-gradient(#35142c,#a63236 65%,#ed813f)",
    land:"#54252b", ground:"#281c28"
  },
  city: {
    sky:"🌆", objects:"🏙️　🚁　🚗",
    background:"linear-gradient(#192b56,#7054a0 60%,#f5a16d)",
    land:"#33344f", ground:"#1a1b30"
  }
};

function snapshot(){
  return {
    charName:$("charName").value,
    outfit:$("outfit").value,
    role:$("role").value,
    power:$("power").value,
    nameInput:$("nameInput").value,
    theme:$("theme").value,
    descInput:$("descInput").value,
    villains:[...villains], extras:[...extras]
  };
}

function remember(){
  history.push(snapshot());
  if(history.length>30)history.shift();
}

function render(){
  const t=themes[$("theme").value] || themes.fantasy;
  $("worldName").textContent=$("nameInput").value || "My Universe";
  $("worldDesc").textContent=$("descInput").value;
  $("world").style.background=t.background;
  $("sky").textContent=t.sky;
  $("sceneObjects").textContent=t.objects;
  $("land").style.background=t.land;
  $("ground").style.background=t.ground;
  $("avatarBody").style.background=$("outfit").value;
  $("powerAura").textContent=
    /fire/i.test($("power").value)?"🔥":
    /ice/i.test($("power").value)?"❄️":
    /water/i.test($("power").value)?"💧":
    /shadow/i.test($("power").value)?"🌑":
    /light/i.test($("power").value)?"⚡":"✨";

  $("characterInfo").textContent=
    ($("charName").value||"Player")+" · "+$("role").value;
  $("powerInfo").textContent="Power: "+$("power").value;
  $("villainInfo").textContent="Villains: "+(villains.join(", ")||"None yet");
  $("extras").textContent=extras.join(" · ")||"Your story starts here.";
}

function updateCharacter(){
  remember();render();message("Character updated successfully!");
}

function applyWorld(){
  remember();render();message("Your world has changed!");
}

function message(text){
  $("log").textContent=text;
  $("status").textContent="● "+text.toUpperCase();
}

function setOutfit(color){
  const colors={
    red:"#ef4444",blue:"#3b82f6",green:"#22c55e",
    black:"#222222",purple:"#a28cff",pink:"#ec70bc",
    white:"#eeeeee",gold:"#f4bd36"
  };
  const found=Object.keys(colors).find(c=>color.toLowerCase().includes(c));
  if(found)$("outfit").value=colors[found];
  else if(/^#[0-9a-f]{6}$/i.test(color))$("outfit").value=color;
}

function runCommand(){
  const raw=$("command").value.trim();
  if(!raw)return;
  const c=raw.toLowerCase();
  remember();

  if(c.includes("temporary")){
    temporary=true;
    savedTemporaryState=history.pop()||snapshot();
    message("Temporary mode enabled. Type 'undo' to revert your last change.");
  }else if(c==="undo"||c.includes("undo change")){
    undoChange(false);
    $("command").value="";
    return;
  }else if(c.includes("make a ")||c.includes("change my world")||
           c.includes("world to")||c.includes("create a world")||
           c.includes("world should be")){
    if(c.includes("space"))$("theme").value="space";
    else if(c.includes("forest"))$("theme").value="forest";
    else if(c.includes("ocean")||c.includes("underwater"))$("theme").value="ocean";
    else if(c.includes("lava")||c.includes("volcano"))$("theme").value="lava";
    else if(c.includes("city")||c.includes("futuristic"))$("theme").value="city";
    else if(c.includes("fantasy")||c.includes("kingdom"))$("theme").value="fantasy";
    $("descInput").value=raw;
    message("World updated! Choose a theme in World Settings for more options.");
  }else if(c.includes("villain")||c.includes("enemy")){
    const m=raw.match(/(?:named|called)\s+(.+)/i);
    const name=m?m[1].replace(/[.!?]+$/,""):"Mystery Villain";
    villains.push(name);
    extras.push("⚔️ "+name+" has entered your story");
    message("New villain created: "+name);
  }else if(c.includes("hero")){
    const m=raw.match(/(?:named|called)\s+(.+)/i);
    const name=m?m[1].replace(/[.!?]+$/,""):"New Hero";
    extras.push("🦸 Hero: "+name);
    message("Hero added to your universe: "+name);
  }else if(c.includes("power")){
    let p=raw.replace(/give me|add|change|my|a|an|the|power|powers|to|with/gi,"").trim();
    $("power").value=p||"Superpower";
    message("New power unlocked: "+$("power").value);
  }else if(c.includes("outfit")||c.includes("clothes")||c.includes("wear")){
    const colors=["red","blue","green","black","purple","pink","white","gold"];
    const found=colors.find(col=>c.includes(col));
    if(found){setOutfit(found);message("Outfit changed to "+found+"!");}
    else message("Choose a colour name, such as red, blue, purple or gold.");
  }else if(c.includes("name me")||c.includes("my name")||
           c.includes("character name")){
    const m=raw.match(/(?:name me|my name is|character name is)\s+(.+)/i);
    if(m){$("charName").value=m[1];message("Character renamed!");}
    else message("Try: My name is Nova");
  }else if(c.includes("dragon")||c.includes("castle")||
           c.includes("car")||c.includes("weapon")||
           c.includes("mission")||c.includes("pet")){
    extras.push("✨ "+raw);
    message("Added to your story: "+raw);
  }else{
    $("descInput").value=raw;
    extras.push("💭 "+raw);
    message("Idea added! I can apply supported commands; for other ideas, edit World Settings or add details to your story.");
  }

  render();
  $("command").value="";
}

function undoChange(showMessage=true){
  if(!history.length){
    if(showMessage)message("Nothing to undo yet.");
    return;
  }
  const s=history.pop();
  $("charName").value=s.charName;
  $("outfit").value=s.outfit;
  $("role").value=s.role;
  $("power").value=s.power;
  $("nameInput").value=s.nameInput;
  $("theme").value=s.theme;
  $("descInput").value=s.descInput;
  villains=s.villains;
  extras=s.extras;
  temporary=false;
  render();
  if(showMessage)message("Last change undone!");
}

function saveWorld(){
  localStorage.setItem("myUniverse",JSON.stringify(snapshot()));
  message("World saved in this browser!");
}

function loadWorld(){
  const raw=localStorage.getItem("myUniverse");
  if(!raw){message("No saved world found. Save your world first.");return;}
  try{
    const s=JSON.parse(raw);
    $("charName").value=s.charName;
    $("outfit").value=s.outfit;
    $("role").value=s.role;
    $("power").value=s.power;
    $("nameInput").value=s.nameInput;
    $("theme").value=s.theme;
    $("descInput").value=s.descInput;
    villains=s.villains||[];
    extras=s.extras||[];
    render();message("Saved world loaded!");
  }catch(e){message("Could not load the saved world.");}
}

function randomize(){
  remember();
  const names=["Nova","Shadow","Blaze","Pixel","Storm"];
  const powers=["Fire","Ice","Lightning","Shadow","Water"];
  $("charName").value=names[Math.floor(Math.random()*names.length)];
  $("power").value=powers[Math.floor(Math.random()*powers.length)];
  $("outfit").value="#"+Math.floor(Math.random()*16777215).toString(16).padStart(6,"0");
  render();message("Random character generated!");
}

function resetWorld(){
  if(!confirm("Reset your current world? Unsaved changes will be lost."))return;
  history=[];
  villains=[];extras=[];temporary=false;
  $("charName").value="Player One";
  $("outfit").value="#a28cff";
  $("role").value="Hero";
  $("power").value="Lightning";
  $("nameInput").value="Crystal Kingdom";
  $("theme").value="fantasy";
  $("descInput").value="A magical kingdom full of secrets.";
  render();message("A fresh universe awaits!");
}

render();
</script>
